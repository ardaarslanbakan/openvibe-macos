cmake_minimum_required(VERSION 3.20)

# Minimal macOS entry point for an OpenViBE SDK-only build.
# The upstream SDK CMakeLists is normally included by a larger, Linux-oriented
# build harness; this wrapper supplies its expected shared variables.
project(OpenViBEMacOSPort LANGUAGES C CXX)

set(CMAKE_MACOSX_RPATH ON)
# The development install uses bin/../lib; the .app package uses
# Contents/MacOS/../Frameworks.  Keep both locations so the same binary works
# in either layout.
set(CMAKE_INSTALL_RPATH "@executable_path/../lib;@executable_path/../Frameworks")

set(OPENVIBE_MAJOR_VERSION 3)
set(OPENVIBE_MINOR_VERSION 2)
set(OPENVIBE_PATCH_VERSION 0)
set(CMAKE_MODULE_PATH_BASE "${CMAKE_CURRENT_SOURCE_DIR}/sdk/cmake-modules")
set(Flag_VerboseOutput ON)
set(DIST_ROOT "${CMAKE_BINARY_DIR}/dist")
set(DIST_BINDIR "${DIST_ROOT}/bin")
set(DIST_LIBDIR "${DIST_ROOT}/lib")
set(DIST_INCLUDEDIR "${DIST_ROOT}/include")
set(DIST_DATADIR "${DIST_ROOT}/share")
set(BUILD_DATADIR "${CMAKE_BINARY_DIR}/data")

add_compile_definitions(
  TARGET_BUILDTYPE_Release
  OV_CMAKE_PATH_ROOT="${DIST_ROOT}"
  OV_CMAKE_PATH_BIN="${DIST_BINDIR}"
  OV_CMAKE_PATH_DATA="${DIST_DATADIR}/openvibe"
  OV_CMAKE_PATH_LIB="${DIST_LIBDIR}"
)

set(KERNEL_FOLDER "Kernel")
set(MISC_FOLDER "Miscellaneous")
set(MODULES_FOLDER "Modules")
set(PLUGINS_FOLDER "Plugins")
set(APP_FOLDER "Applications")
set(TESTS_FOLDER "Tests")
set(VALIDATION_FOLDER "Validation")
